       ```python
       @router.post("/notifications/panel/mark-all-read", response_class=HTMLResponse)
       async def mark_all_panel_notifications_read(
           request: Request,
           lang: str | None = Query(default=None),
           session: Session = Depends(get_fastapi_session),
       ):
           """标记所有未读通知为已读，返回更新后的面板内容和 Badge OOB。"""
           current_lang = detect_lang(request, lang)
           t = get_translations(current_lang)
           
           repo = NotificationRepository()
           count = repo.mark_all_as_read()
           
           # 构建响应：面板内容 + Badge OOB
           context = build_notification_panel_context(
               session=session,
               lang=current_lang,
               page=0,
               page_size=10,
           )
           context["request"] = request
           context["has_unread"] = False  # 全部已读
           
           templates = get_templates()
           response = templates.TemplateResponse("partials/notification_panel_content.html", context)
           
           # 添加 Badge OOB 更新（空字符串 = 隐藏 badge）
           # 通过 HX-Trigger 通知前端更新 badge
           response.headers["HX-Trigger"] = json.dumps(
               {
                   "notificationsUpdated": True,
                   "showToast": {"message": t.get("notification_marked_read", "已标记 {count} 条通知为已读").format(count=count)}
               },
               ensure_ascii=True
           )
           
           return response
       ```

    - 通过 HX-Trigger 触发 badge 刷新而非 OOB swap（更简洁）
